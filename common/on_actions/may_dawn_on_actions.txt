# PLC Startup Fix
on_game_started = {
    on_actions = {
        may_dawn_on_game_started
    }
}

# Osobna on_action – vanilla ma już własny blok effect w on_game_started
may_dawn_on_game_started = {
    effect = {
        if = {
            limit = {
                exists = c:PLC
            }
            c:PLC = {
                add_modifier = {
                    name = modifier_post_war_reality
                    years = 10
                    is_decaying = yes
                }
            }
        }
    }
}
