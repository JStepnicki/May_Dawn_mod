# --- PLC ANNEXATION ON ACTIONS ---
# Historia stanów z vanilli tworzy w tych regionach stany Rosji, Polski (POL), Austrii i Prus,
# a historia moda zabiera im prowincje. Zostają wtedy resztki albo puste "split state" bez terytorium.
# Zamiast przenosić pojedyncze prowincje (co psuło się, gdy patch zmieniał mapę),
# przekazujemy PLC/COU każdy obcy stan w regionie – gra scala je w jeden stan.

on_game_started = {
    on_actions = {
        plc_history_setup_on_action
    }
}

plc_history_setup_on_action = {
    effect = {
        # --- Regiony w całości należące do Rzeczypospolitej ---

            every_scope_state = {
                limit = { NOT = { owner = c:PLC } }
                set_state_owner = c:PLC
            }

            }

            }
        }

            }
        }
        s:STATE_WEST_PRUSSIA = {

            }
        }
        s:STATE_WEST_GALICIA = {
